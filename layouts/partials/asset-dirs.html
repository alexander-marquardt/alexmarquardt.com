{{- /* The image directories directly under a page's own, with the number of images in each. Takes the page; returns a slice of {name, label, pagehref, asseturl, assetdir, count}. This walks the filesystem rather than reading the page's children, for the same reason the file listings do: the directory tree is the one source of truth about what exists. It also does not depend on how Hugo classifies this page -- the site's taxonomy configuration makes every top-level section a `term` on some Hugo versions and a `section` on others, and a listing built from `.Pages` silently comes out EMPTY on the former. */ -}} {{- $page := . -}} {{- $config := site.Params.demoAssets -}} {{- $dir := $page.Params.assetdir | default $config.root -}} {{- $base := $page.Params.asseturl | default $config.url -}} {{- $dirs := slice -}} {{- range os.ReadDir $dir -}} {{- if .IsDir -}} {{- $name := .Name -}} {{- $child := printf "%s/%s" $dir $name -}} {{- $count := partial "asset-count.html" $child -}} {{- $dirs = $dirs | append (dict "name" $name "label" ($name | replaceRE "[-_]" " " | title) "pagehref" (printf "%s%s/" $page.RelPermalink $name) "asseturl" (printf "%s/%s" $base $name) "assetdir" $child "count" $count) -}} {{- end -}} {{- end -}} {{- return $dirs -}}

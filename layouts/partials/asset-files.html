{{- /* Every image in one directory, read at build time. Takes a dict: dir the directory to read base the URL those files are served from label optional; the directory name, for a grid spanning several labelhref optional; where that label links Returns a slice of {href, name, size, label, labelhref}. Nothing is listed in content: this reads the directory, so an image added to it appears on the next build with no edit anywhere. */ -}} {{- $dir := .dir -}} {{- $base := .base -}} {{- $label := .label | default "" -}} {{- $labelhref := .labelhref | default "" -}} {{- $extensions := slice ".png" ".jpg" ".jpeg" ".gif" ".webp" ".svg" -}} {{- $items := slice -}} {{- range os.ReadDir $dir -}} {{- if and (not .IsDir) (in $extensions (path.Ext .Name | lower)) -}} {{- $items = $items | append (dict "href" (printf "%s/%s" $base .Name) "name" .Name "size" .Size "label" $label "labelhref" $labelhref) -}} {{- end -}} {{- end -}} {{- return $items -}}
